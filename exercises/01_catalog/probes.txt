#[cfg(test)]
mod workshop_test_support {
    pub struct Cwd { previous: std::path::PathBuf, pub dir: tempfile::TempDir }
    impl Drop for Cwd { fn drop(&mut self) { std::env::set_current_dir(&self.previous).unwrap(); } }
    pub fn cwd(with_cargo: bool) -> Cwd {
        let previous=std::env::current_dir().unwrap(); let dir=tempfile::tempdir().unwrap();
        if with_cargo {
            std::fs::create_dir_all(dir.path().join("dev")).unwrap();
            std::fs::create_dir_all(dir.path().join("exercises")).unwrap();
            let mut bins=String::new();
            for (name,source) in [
                ("probe_ok","fn main() {} #[test] fn works() { assert_eq!(2+2,4); }"),
                ("probe_ok_sol","fn main() {} #[test] fn works() { assert_eq!(2+2,4); }"),
                ("probe_runtime","fn main() { std::process::exit(7); }"),
                ("probe_tests","fn main() {} #[test] fn broken() { panic!(\"contract failure\"); }"),
                ("probe_compile","fn main() { missing_function(); }"),
            ] {
                std::fs::write(dir.path().join(format!("exercises/{name}.rs")),source).unwrap();
                bins.push_str(&format!("[[bin]]\nname = {name:?}\npath = \"../exercises/{name}.rs\"\n"));
            }
            let manifest=format!("[package]\nname = \"workshop_fixture\"\nversion = \"0.0.0\"\nedition = \"2024\"\n{bins}");
            std::fs::write(dir.path().join("dev/Cargo.toml"),&manifest).unwrap();
            std::fs::write(dir.path().join("Cargo.toml"),manifest.replace("../exercises/","exercises/")).unwrap();
            std::fs::write(dir.path().join("info.toml"),"format_version = 1\n[[exercises]]\nname=\"probe_ok\"\nhint=\"fixture\"\n").unwrap();
        }
        std::env::set_current_dir(dir.path()).unwrap(); Cwd { previous, dir }
    }
    pub fn info(name: &'static str) -> crate::info_file::ExerciseInfo<'static> {
        crate::info_file::ExerciseInfo {name,dir:None,test:true,strict_clippy:false,input_files:vec![],hint:"fixture".into(),skip_check_unsolved:false}
    }
    pub fn app() -> (Cwd,crate::app_state::AppState) {
        let cwd=cwd(true);
        let (state,_)=crate::app_state::AppState::new(vec![info("alpha"),info("beta"),info("gamma")],String::new(),None,false).unwrap();
        (cwd,state)
    }
}
