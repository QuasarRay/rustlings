        pub static FINAL_KEY: std::sync::atomic::AtomicU8 = std::sync::atomic::AtomicU8::new(0);
        pub fn read()->std::io::Result<Event> {
            use std::sync::atomic::Ordering::Relaxed;

            Err(std::io::Error::other("script exhausted"))
        }
    }
